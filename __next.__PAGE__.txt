window.initFallbackTripMap = function initFallbackTripMap() {
  const mapEl = document.getElementById("trip-map");
  if (!mapEl || mapEl.dataset.ready === "leaflet" || !window.L) return;
  mapEl.dataset.ready = "leaflet";
  mapEl.innerHTML = "";

  const map = L.map("trip-map", {
    scrollWheelZoom: false,
    touchZoom: true,
    doubleClickZoom: true,
    dragging: true,
    tap: false,
    zoomSnap: 0.25,
    zoomDelta: 0.5,
    zoomControl: true,
    attributionControl: false
  }).setView([-33.6, 20.75], 6);

  window.tripMapBridge = {
    zoomIn: function () { map.zoomIn(); },
    zoomOut: function () { map.zoomOut(); },
    getZoom: function () { return map.getZoom(); },
    getCenter: function () {
      const center = map.getCenter();
      return { lat: center.lat, lng: center.lng };
    }
  };

  L.tileLayer("https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}").addTo(map);

  const mainOut = [[-33.9249, 18.4241], [-34.4092, 19.2504], [-34.0226, 20.4417], [-34.0527, 23.3716]];
  const returnLine = [[-34.0527, 23.3716], [-33.5907, 22.2014], [-33.9619, 20.7198], [-33.7342, 18.9621], [-32.7252, 17.9502], [-33.9249, 18.4241]];
  const addoLine = [[-34.0527, 23.3716], [-33.9611, 25.6022], [-33.4837, 25.7501]];

  L.polyline(mainOut, { color: "#f2b705", weight: 4 }).addTo(map);
  L.polyline(returnLine, { color: "#f05a4f", weight: 4 }).addTo(map);
  L.polyline(addoLine, { color: "#56a36c", weight: 4, dashArray: "8 8" }).addTo(map);

  const bounds = [];
  window.tripStops.forEach((stop) => {
    const latLng = [stop.position.lat, stop.position.lng];
    bounds.push(latLng);
    L.circleMarker(latLng, {
      radius: stop.major ? 8 : 5,
      color: "#ffffff",
      weight: stop.major ? 2.5 : 1.8,
      fillColor: stop.optional ? "#56a36c" : "#f05a4f",
      fillOpacity: stop.major ? 0.96 : 0.82
    })
      .addTo(map)
      .bindTooltip(stop.label, {
        permanent: true,
        direction: stop.direction || "top",
        offset: stop.offset || [0, -8],
        className: stop.major ? "stopLabel majorStopLabel" : "stopLabel minorStopLabel"
      });
  });

  setTimeout(function () { map.invalidateSize(); }, 200);
};

window.startFallbackTripMap = function startFallbackTripMap() {
  let tries = 0;
  const timer = setInterval(function () {
    if (window.L) {
      clearInterval(timer);
      window.initFallbackTripMap();
    }
    tries += 1;
    if (tries > 24) clearInterval(timer);
  }, 250);
};

window.addEventListener("click", function (event) {
  const button = event.target.closest("[data-map-zoom]");
  if (!button || !window.tripMapBridge) return;
  if (button.dataset.mapZoom === "in") window.tripMapBridge.zoomIn();
  if (button.dataset.mapZoom === "out") window.tripMapBridge.zoomOut();
});

window.addEventListener("DOMContentLoaded", function () {
  window.startFallbackTripMap();
